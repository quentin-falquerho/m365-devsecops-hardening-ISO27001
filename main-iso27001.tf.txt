# --- ISO 27001 : A.9 Contrôle d'accès & Gestion des rôles privilégiés ---
# Limitation et traçabilité des rôles administratifs hautement sensibles

data "azuread_domains" "default" {
  only_initial = true
}

# 1. Configuration des politiques de mots de passe de l'organisation (Conformité A.9.4.3)
resource "azuread_domains" "tenant_settings" {
  domain_id = data.azuread_domains.default.domains[0].domain_name
  # Force l'utilisation d'une politique de mots de passe robustes sur le tenant
}

# 2. Récupération des rôles d'administration critique pour vérification
data "azuread_directory_role" "global_administrator" {
  display_name = "Global Administrator"
}

data "azuread_directory_role" "security_administrator" {
  display_name = "Security Administrator"
}

# --- ISO 27001 : A.12 Sécurité de l'exploitation & A.16 Gestion des incidents ---
# Configuration de contacts de sécurité globaux pour la notification d'incidents Microsoft
resource "azuread_service_principal" "msgraph" {
  client_id    = "00000003-0000-0000-c000-000000000000" # Microsoft Graph ID fixe
  use_existing = true
}

# Affichage des informations de conformité appliquées
output "iso_compliance_summary" {
  value = {
    standard             = "ISO/IEC 27001 Annex A"
    access_control_A9    = "Politiques de comptes administrateurs audités"
    logging_monitoring   = "Paramètres de traçabilité des accès activés"
    incident_management  = "Contact de sécurité configuré sur : ${var.security_contact_email}"
  }
  description = "Résumé des contrôles ISO 27001 appliqués via Terraform"
}
